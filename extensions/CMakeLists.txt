# client_app is declared before this directory; feature hosts are attached there
# without changing the existing native endpoint/plugin ABI.

option(ROBOMASTER_BUILD_BATTLEFIELD "Build optional display-only situation platform" ON)
option(ROBOMASTER_BUILD_3D_MAP "Install the optional lazy-loaded GLB map renderer" ON)

    add_subdirectory(battlefield-minimap)
endif()
if(ROBOMASTER_BUILD_3D_MAP AND IS_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}/battlefield-3d")
    add_subdirectory(battlefield-3d)
endif()

if(ROBOMASTER_BUILD_BATTLEFIELD)
    add_subdirectory(battlefield)
endif()

if(ROBOMASTER_BUILD_HERO_LOB_VIDEO)
    add_subdirectory(hero-lob-video)
endif()

option(ROBOMASTER_BUILD_LAB_EXTENSIONS "Install optional season audit, source health and tactical timeline" ON)
if(ROBOMASTER_BUILD_LAB_EXTENSIONS)
    foreach(_feature season-audit source-health tactical-timeline)
        if(IS_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}/${_feature}")
            add_custom_target(${_feature}_assets ALL
                COMMAND ${CMAKE_COMMAND} -E copy_directory "${CMAKE_CURRENT_SOURCE_DIR}/${_feature}" "${CMAKE_BINARY_DIR}/extensions/${_feature}")
            install(DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}/${_feature}" DESTINATION share/robomaster/extensions PATTERN "__pycache__" EXCLUDE)
        endif()
    endforeach()
endif()
